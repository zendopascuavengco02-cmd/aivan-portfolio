My Java Project

CC102 FINAL PERFORMANCE TASK

Number Guessing Game

This Java console-based game generates a random number from 1 to 20. The player tries to guess the number. It demonstrates the use of variables, Scanner, Random, conditional statements, and a do-while loop.

Java Source Code

import java.util.Scanner;
import java.util.Random;

public class NumberGuessingGame {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int secretNumber = random.nextInt(20) + 1;
        int guess;
        int attempts = 0;

        System.out.println("==============================");
        System.out.println("     NUMBER GUESSING GAME");
        System.out.println("==============================");
        System.out.println("Guess the number from 1 to 20!");

        do {
            System.out.print("\nEnter your guess: ");

            if (input.hasNextInt()) {
                guess = input.nextInt();
                attempts++;

                if (guess < 1 || guess > 20) {
                    System.out.println(
                        "Please enter a number from 1 to 20."
                    );
                } else if (guess < secretNumber) {
                    System.out.println("Too low! Try again.");
                } else if (guess > secretNumber) {
                    System.out.println("Too high! Try again.");
                } else {
                    System.out.println(
                        "Congratulations! You guessed it!"
                    );
                    System.out.println(
                        "Number of attempts: " + attempts
                    );
                }
            } else {
                System.out.println(
                    "Invalid input! Enter a number."
                );
                input.next();
                guess = 0;
            }

        } while (guess != secretNumber);

        System.out.println("\nThank you for playing!");
        input.close();
    }
}
